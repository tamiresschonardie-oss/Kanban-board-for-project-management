3️⃣ Estado Inicial com useEffect (Sem Reset)
// Todas as fases começam expandidas
const [expandedPhases, setExpandedPhases] = useState<Set<string>>(
  () => new Set(project.phases?.map(p => p.id) || [])
  //   ↑ Factory function (executa UMA VEZ na primeira render)
);
// Toggle preserva state (não reseta)
const togglePhase = (phaseId: string) => {
  setExpandedPhases(prev => {
    const newSet = new Set(prev);
    if (newSet.has(phaseId)) {
      newSet.delete(phaseId);
    } else {
      newSet.add(phaseId);
    }
    return newSet;  // ← Novo Set com mudança localizada
  });
};
✅ Expandidas por padrão (primeira render)
✅ Sem reset ao interagir (toggle mantém state)
✅ Sem useEffect (state é responsável)
